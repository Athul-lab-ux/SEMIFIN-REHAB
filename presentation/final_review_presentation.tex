\begin{frame}{07 · Mathematical Foundation: Real-Time Layer (Part 1)}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{$C_1$ · Joint Goniometry (Elbow \& Shoulder)}
$$\theta = \arccos\left(\frac{\vec{a} \cdot \vec{b}}{\|\vec{a}\| \|\vec{b}\|}\right) \times \frac{180^\circ}{\pi}$$
\begin{itemize}
    \item $\vec{a} = \vec{P}_{\text{sh}} - \vec{P}_{\text{el}}$, $\vec{b} = \vec{P}_{\text{wr}} - \vec{P}_{\text{el}}$.
    \item Evaluates real-time elbow angle for extension ($\ge 140^\circ$) and flexion return ($< 125^\circ$).
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{$C_2$ · Target Interception Collision}
$$d = \sqrt{(\Delta x)^2 + (\Delta y)^2} \le r_{\text{target}}$$
\begin{itemize}
    \item Evaluates when patient's wrist or fingertip reaches target.
    \item Drives collision detection in arcade arena and AR targets.
\end{itemize}
\end{block}
\end{columns}

\vspace{2mm}
\begin{block}{$C_3$ · Scale-Invariant Palm Dispersion (Fist vs. Spread)}
$$\text{Spread Ratio} = \frac{\frac{1}{5}\sum_{i \in \{4,8,12,16,20\}} \|\vec{P}_i - \vec{P}_{\text{palmCenter}}\|}{\|\vec{P}_9 - \vec{P}_0\|}$$
\begin{itemize}
    \item Normalized against palm scale ($\|\vec{P}_9 - \vec{P}_0\|$) for size-invariant tracking.
    \item Deterministic gating: $\text{Ratio} \le 0.95 \implies \text{Closed Fist ✊}$; $\text{Ratio} \ge 1.05 \implies \text{Open Palm 🖐️}$.
\end{itemize}
\end{block}
\end{frame}

% ==================== SLIDE 11 ====================
\begin{frame}{08 · Mathematical Foundation: Real-Time Layer (Part 2)}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{$E_1$ · Anti-Cheat Trunk Lateral Tilt Angle}
$$\text{Tilt} = \left|\arctan2(\Delta y_{\text{sh}}, \Delta x_{\text{sh}})\right| \times \frac{180^\circ}{\pi}$$
\begin{itemize}
    \item Evaluates inter-shoulder line slope.
    \item If $\text{Tilt} > 25^\circ$, session locks out and buzzes: \textit{"Posture cheat detected: keep shoulders level."}
    \item Contextually exempted for shoulder shrugs.
\end{itemize}
\end{alertblock}

\column{0.48\textwidth}
\begin{block}{$E_2$ · Scale-Invariant Pincer Grasp Ratio}
$$\text{Pinch Ratio} = \frac{\|\vec{P}_4 - \vec{P}_8\|}{\|\vec{P}_9 - \vec{P}_0\|}$$
\begin{itemize}
    \item Distance between thumb tip (4) and index tip (8).
    \item Step 1: Ratio $\le 0.32$ (Pinch locked 🤏).
    \item Step 2: Ratio $\ge 0.40$ (Release wide 🖐️).
    \item Independent of patient hand size.
\end{itemize}
\end{block}
\end{columns}

\vspace{2mm}
\begin{block}{$C_4$ · Fast Fourier Transform (Tremor Isolation)}
$$X[k] = \sum_{n=0}^{N-1} x[n] \cdot e^{-j \frac{2\pi kn}{N}}$$
\begin{itemize}
    \item Converts wrist position buffer time-series into frequency domain.
    \item Detects pathological frequency spikes in the clinical 4–12 Hz tremor spectrum.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{09 · Analytical \& Clinical Math Extensions}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{$S_1$ · Movement Smoothness Index (0–100)}
$$\text{Score} = 100 - \min\left(100, 20\log_{10}\left(\frac{\frac{1}{2}\int \text{Jerk}^2 dt \cdot T^5}{D^2}\right)\right)$$
\begin{itemize}
    \item Jerk computed via discrete finite differences.
    \item Normalizes dimensionless jerk into 0–100 clinical score.
\end{itemize}
\end{block}

\begin{block}{$E_3$ · Signed Wrist Pitch Angle}
$$\text{Pitch} = \arctan2(\Delta y_{\text{hand}}, \Delta x_{\text{hand}}) \times \frac{180^\circ}{\pi}$$
\begin{itemize}
    \item Upward cocking $\ge 14^\circ \rightarrow$ level return $\le 10^\circ$.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{$P_1$ · Harmonic Pursuit Trajectory}
$$\begin{cases} X_{\text{orb}}(t) = 0.50 + 0.28\sin(0.65t) \\ Y_{\text{orb}}(t) = 0.48 + 0.08\sin(1.30t) \end{cases}$$
\begin{itemize}
    \item Generates smooth sinusoidal gliding target orb.
    \item Live reverse countdown; resets if wrist leaves $>450$ ms.
\end{itemize}
\end{block}

\begin{block}{$B_1$ · Auditory Biofeedback Sonification}
$$f(\theta) = f_{\min} + \left(\frac{\theta - \theta_{\min}}{\theta_{\max} - \theta_{\min}}\right)(f_{\max} - f_{\min})$$
\begin{itemize}
    \item Live ROM mapped continuously from 220 Hz to 880 Hz.
\end{itemize}
\end{block}
\end{columns}
\end{frame}
